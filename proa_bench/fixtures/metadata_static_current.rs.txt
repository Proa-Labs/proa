use proa_core::text::StaticRaw;
use proa_core::{DynLoaderHandle, Metadata, WebContext, WebRenderSync, WriteError};
use proa_framework_axum::{router_from_spec, FrameworkBuilder, Layout, Route, RouteResponse};

fn defaults() -> Metadata {
    Metadata::new()
        .title("Store")
        .title_template("%s | Store")
        .description("Plans")
        .open_graph(|og| {
            og.site_name("Store");
            og.ty("website");
            og.url("https://example.com/pricing");
        })
}
struct Site;
impl Layout for Site {
    fn setup(&self, cx: &mut WebContext<DynLoaderHandle>) -> Result<(), WriteError> {
        cx.metadata(defaults())?;
        Ok(())
    }
}
struct Document;
impl WebRenderSync for Document {
    fn render(self, cx: &mut WebContext) -> Result<(), WriteError> {
        StaticRaw("<html><head>").render(cx)?;
        cx.metadata.head().render(cx)?;
        StaticRaw("</head><body>Pricing</body></html>").render(cx)
    }
}
pub fn app() -> axum::Router {
    let (routes, setups) = router_from_spec(vec![Route::layout(
        "/",
        &Site,
        vec![Route::page(
            "pricing",
            "Pricing",
            axum::routing::get(|| async { RouteResponse::ssr(Document) }),
        )],
    )]);
    FrameworkBuilder::new(manifest())
        .with_dynamic_routes_and_setups(routes, setups)
        .build()
}
fn manifest() -> proa_framework_axum::IslandManifest {
    proa_framework_axum::IslandManifest {
        islands: Default::default(),
        assets: Default::default(),
        import_map: Default::default(),
        ssr_bundle: None,
    }
}
pub struct Prepared(Metadata);
impl Prepared {
    pub fn new() -> Self {
        Self(defaults())
    }
    pub fn emit(&self, cx: &mut WebContext) {
        cx.metadata(self.0.clone()).unwrap();
        cx.__next_metadata_route_scope().unwrap();
        cx.metadata(Metadata::new().title("Pricing")).unwrap();
        cx.metadata.head().render(cx).unwrap();
    }
}
